cmake_minimum_required(VERSION 3.30)
project(8ball)

add_executable(eightball)

target_compile_features(eightball PRIVATE cxx_std_20)

set_target_properties(eightball PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
)

target_sources(eightball
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/PoolTask.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ThreadPool.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ThreadWorker.cpp
)

target_include_directories(eightball
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/include
)
